#defining the non impact means and percent differences from impact vars
pct_diff <- function(x, ref) {
  ((x - ref) / ref) * 100
}

policy_summary <- function(df, var) {

  no_impact_mean <- mean(
    df[[var]][df$impact == 0],
    na.rm = TRUE
  )

  impact_mean <- mean(
    df[[var]][df$impact == 1],
    na.rm = TRUE
  )

  direct_mean <- mean(
    df[[var]][df$incarc == 1],
    na.rm = TRUE
  )

  proximal_mean <- mean(
    df[[var]][df$knowincarc == 1],
    na.rm = TRUE
  )

  victim_mean <- mean(
    df[[var]][df$vict == 1],
    na.rm = TRUE
  )

  tibble::tibble(
    var = var,
    no_impact_mean = no_impact_mean,
    impact_diff = mean_diff(impact_mean, no_impact_mean),
    direct_diff = mean_diff(direct_mean, no_impact_mean),
    proximal_diff = mean_diff(proximal_mean, no_impact_mean),
    victim_diff = mean_diff(victim_mean, no_impact_mean)
  )
}

pval_ttest <- function(df, var, group_var) {
  sub <- df %>%
    dplyr::filter(!is.na(.data[[var]]), !is.na(.data[[group_var]])) %>%
    dplyr::filter(.data[[group_var]] %in% c(0, 1))

  if (length(unique(sub[[group_var]])) < 2) return(NA_real_)

  stats::t.test(sub[[var]] ~ sub[[group_var]])$p.value
}

mean_diff <- function(x, ref) {
  x - ref
}

policy_summary <- function(df, var) {
  no_impact_mean <- mean(df[[var]][df$impact == 0], na.rm = TRUE)
  impact_mean <- mean(df[[var]][df$impact == 1], na.rm = TRUE)
  direct_mean    <- mean(df[[var]][df$incarc == 1], na.rm = TRUE)
  proximal_mean  <- mean(df[[var]][df$knowincarc == 1], na.rm = TRUE)
  victim_mean    <- mean(df[[var]][df$vict == 1], na.rm = TRUE)
  
  tibble::tibble(
    var = var,
    no_impact_mean = no_impact_mean,
    impact_mean = pct_diff(impact_mean, no_impact_mean),
    direct_pct = pct_diff(direct_mean, no_impact_mean),
    proximal_pct = pct_diff(proximal_mean, no_impact_mean),
    victim_pct = pct_diff(victim_mean, no_impact_mean)
  )
}
